# 三阶线的典型方程和图形

1. 三次多项式函数： $y = ax^3 + bx^2 + cx + d$
2. 椭圆曲线（一般型）： $y^2 = x^3 + ax + b$
3. 退化椭圆曲线（可分解）： $y^2 = x^2 (x + 1)$
4. 椭圆曲线（缺线性项）： $y^2 = x^3 + k$
5. 椭圆曲线（常数项 1）： $y^2 = x^3 + 1$
6. 椭圆曲线（含线性项）： $y^2 = x^3 - x$
7. 椭圆曲线（含系数）： $y^2 = 4x^3 - \frac{1}{27}$
8. 退化椭圆曲线（可分解）： $y^2 = x^3 - x^2$
9. 退化椭圆曲线（尖点）： $y^2 = x^3$
10. （与第 3 条相同） $y^2 = x^3 + x^2$
11. 三次隐式曲线： $x^3 + y^3 = x^2 + y^2$
12. 笛卡尔叶形线： $y^3 + x^3 - 3xy = 0$
13. 三次有理曲线： $xy = 1 - x - x^2 - x^3$
14. 有理三次曲线（含参数）： $y(x^2 + a^2) = a^3$
15. 三次隐式曲线： $2x^2y + y^2 = x$

### 绘出隐式方程的图像

```python
import numpy as np
import matplotlib.pyplot as plt

# 定义函数
def F(x, y):
    return y**2 - x**3 + x
    # return y**2 - x**2*(x + 1)
    # return y**2 - x**3 - 1
    # return y**2 - x**3 + x    
    # return y**2 - 4*x**3 + 1/27
    # return y**2 - x**3 + x**2
    # return y**2 - x**3
    # return y**2 - x**3 - x**2
    # return x**3 + y**3 - x**2 - y**2
    # return y**3 + x**3 - 3*x*y
    # return xy- 1 + x + x**2 + x**3
    # return xy- 1 + x + x**2 + x**3
    # return 2*x**2*y + y**2 - x
    # return y(x**2 + a**2) - a**3
    # 下面多项式中的系数a,b,c,d,k需要赋值
    # return y - a*x**3 - b*x**2 - c*x - d
    # return y**2 - x**3 - a*x - b
    # return y**2 - x**3 - k

# 创建网格
x = np.linspace(-5, 5, 400)
y = np.linspace(-5, 5, 400)
X, Y = np.meshgrid(x, y)
Z = F(X, Y)

# 画等高线 F=0
plt.figure(figsize=(8, 6))
plt.contour(X, Y, Z, levels=[0], colors='blue', linewidths=2)
plt.axhline(0, color='black', linewidth=0.5)
plt.axvline(0, color='black', linewidth=0.5)
plt.xlabel('x')
plt.ylabel('y')
plt.title('Curve: $y^3 + xy^2 + x^2 y + 6 = 0$')
plt.grid()
plt.axis('equal')
plt.show()
```
### 绘出显式方程的图像
```python
import numpy as np
import matplotlib.pyplot as plt

# 定义函数
def f(x):
    return 2 * x**3 - 3 * x**2 - 5 * x - 7

# 生成 x 数据（从 -3 到 4，共 1000 个点，保证曲线平滑）
x = np.linspace(-3, 4, 1000)
y = f(x)

# 创建图形
plt.figure(figsize=(10, 6))

# 绘制曲线（蓝色，线宽2）
plt.plot(x, y, color='blue', linewidth=2, label=r'$y = 2x^3 - 3x^2 - 5x - 7$')

# 添加坐标轴参考线
plt.axhline(0, color='black', linewidth=0.8, linestyle='--')   # x轴
plt.axvline(0, color='black', linewidth=0.8, linestyle='--')   # y轴

# 设置坐标轴标签、标题和网格
plt.xlabel('x', fontsize=12)
plt.ylabel('y', fontsize=12)
plt.title('Plot of $y = 2x^3 - 3x^2 - 5x - 7$', fontsize=14)
plt.grid(True, linestyle=':', alpha=0.7)

# 显示图例
plt.legend(fontsize=12)

# 设置坐标轴范围（可选，展示完整曲线）
plt.xlim(-3, 4)
plt.ylim(-20, 20)

# 显示图形
plt.show()
```
